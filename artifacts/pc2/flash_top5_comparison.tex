\medskip
\noindent\textbf{Table A --- column guide.}
\begin{description}
  \item[GT] Ground-truth HLS latency (reference gold design per kernel).
  \item[lat/GT ratio] $\text{synth\_latency} / \text{GT\_latency}$; $<1$ faster than GT, $>1$ slower.
  \item[\textbf{\#}] Rank by geo-mean lat/GT (1 = best).
  \item[\textbf{Geo}] Geometric mean of per-bench lat/GT ratios; \textbf{lower is better}.
  \item[\textbf{Spd}] Geo-mean speedup $= 1/\textbf{Geo} = \text{geo-mean}(\text{GT}/\text{synth})$; \textbf{higher is better}.
  \item[\textbf{Mean}] Arithmetic mean of per-bench ratios.
  \item[\textbf{F$<$GT} / \textbf{F$>$GT} / \textbf{Tie}] Kernel counts with ratio $<1$, $>1$, or $\approx 1$ vs ground truth.
  \item[\textbf{Abs} / \textbf{Sh.} / \textbf{Pl.}] Win metrics \textbf{among these 5 modes only} (not vs GT): sole lowest synth latency (Abs); tied for lowest (Sh.); pooled score $=\text{Abs}+\sum 1/n$ on ties (Pl., max.\ 27).
  \item[\textbf{Sk} / \textbf{Inj}] Skills in library file / skills injected into the prompt (U280-filtered).
  \item[old / new] \textbf{old} = \texttt{skills.json} (55). \textbf{new} = frozen ii-target-miss libraries:
    \texttt{skills\_ii\_target\_miss\_solutions\_added(90skills).json} (variant~2, stamp \texttt{20260623\_024548});
    \texttt{skills\_ii\_target\_miss\_solutions\_added(73skills).json} (variant~3, stamp \texttt{20260621\_075846}).
\end{description}

\subsection*{Key takeaways}

\begin{itemize}
  \item \textbf{No avoids (old)} has the best geo-mean vs GT (0.034) and never exceeds GT latency.
  \item \textbf{All+avoids (new)} ranks 2nd on geo-mean, wins the most top-5 pooled benches (6.3/27), and is 14--12 vs no avoids (old) head-to-head.
  \item \textbf{Noskills (old)} is a strong GT baseline (25/27 faster than GT) but wins fewer top-5 bench latencies (4.0 pooled).
  \item Skills library size does not predict rank: 0-skill noskills beats 90-skill all+avoids (old) on geo-mean.
\end{itemize}
