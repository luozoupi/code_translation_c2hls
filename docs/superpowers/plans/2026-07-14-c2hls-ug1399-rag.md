**Goal:** Add opt-in UG1399 documentation RAG to c2hls (`--rag` / `--rag-mode`), with a bundled BM25 index under `artifacts/rag/ug1399`, injecting retrieved context into flashopt / repair / dataflow prompts per mode without changing default (RAG-off) behavior.

**Architecture:** New `c2hls_rag.py` owns config resolution, chunk load, BM25 retrieve, and `should_inject(mode, stage)`. A one-shot `scripts/build_ug1399_rag_index.py` builds the cache. **Tech Stack:** Python 3, stdlib (+ existing pytest), argparse/env in `c2hls.py` and `post_flash_dataflow` entrypoints, BM25 implemented in-repo.

```python
# c2hls_rag.py — key API surface
from __future__ import annotations

import json
import math
import os
import re
from dataclasses import dataclass
from pathlib import Path
from typing import Literal, Optional

RAG_MODES = ("flashopt", "repair", "both", "everywhere")
RagMode = Literal["flashopt", "repair", "both", "everywhere"]
DEFAULT_CORPUS = Path(__file__).resolve().parent / "artifacts" / "rag" / "ug1399"
DEFAULT_TOP_K = 4
CHUNK_SIZE = 1000
OVERLAP = 200

_TOKEN = re.compile(r"[a-z0-9_]+", re.I)


@dataclass(frozen=True)
class RagConfig:
    enabled: bool
    mode: Optional[RagMode]
    corpus_dir: Path
    top_k: int


@dataclass
class RagIndex:
    meta: dict
    chunk_ids: list[str]
    texts: list[str]
    # bm25 stats
    df: dict[str, int]
    tf: list[dict[str, int]]
    doc_len: list[int]
    avgdl: float
    N: int


def chunk_text(text: str, chunk_size: int = CHUNK_SIZE, overlap: int = OVERLAP) -> list[str]:
    text = text.replace("\r\n", "\n")
    if chunk_size <= 0:
        raise ValueError("chunk_size must be positive")
    if overlap < 0 or overlap >= chunk_size:
        raise ValueError("overlap must be in [0, chunk_size)")
    if not text:
        return []
    out: list[str] = []
    step = chunk_size - overlap
    i = 0
    n = len(text)
    while i < n:
        out.append(text[i : i + chunk_size])
        if i + chunk_size >= n:
            break
        i += step
    return out


def tokenize(text: str) -> list[str]:
    return [t.lower() for t in _TOKEN.findall(text)]


def _build_bm25(texts: list[str]) -> tuple[dict[str, int], list[dict[str, int]], list[int], float]:
    tf_list: list[dict[str, int]] = []
    df: dict[str, int] = {}
    doc_len: list[int] = []
    for text in texts:
        toks = tokenize(text)
        doc_len.append(len(toks) or 1)
        tf: dict[str, int] = {}
        for t in toks:
            tf[t] = tf.get(t, 0) + 1
        for t in tf:
            df[t] = df.get(t, 0) + 1
        tf_list.append(tf)
    avgdl = sum(doc_len) / max(len(doc_len), 1)
    return df, tf_list, doc_len, avgdl


def load_index(corpus_dir: Path) -> RagIndex:
    corpus_dir = Path(corpus_dir)
    meta_path = corpus_dir / "index_meta.json"
    chunks_path = corpus_dir / "chunks.jsonl"
    if not meta_path.is_file() or not chunks_path.is_file():
        raise FileNotFoundError(
            f"RAG index missing under {corpus_dir}. "
            f"Build with: python3 scripts/build_ug1399_rag_index.py --source <UG1399> --out {corpus_dir}"
        )
    meta = json.loads(meta_path.read_text(encoding="utf-8"))
    ids: list[str] = []
    texts: list[str] = []
    with chunks_path.open(encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            obj = json.loads(line)
            ids.append(str(obj["id"]))
            texts.append(str(obj["text"]))
    df, tf, doc_len, avgdl = _build_bm25(texts)
    return RagIndex(
        meta=meta,
        chunk_ids=ids,
        texts=texts,
        df=df,
        tf=tf,
        doc_len=doc_len,
        avgdl=avgdl,
        N=len(texts),
    )


def _bm25_score(index: RagIndex, query_tf: dict[str, int], doc_i: int, k1: float = 1.5, b: float = 0.75) -> float:
    score = 0.0
    dl = index.doc_len[doc_i]
    for term, qtf in query_tf.items():
        if term not in index.tf[doc_i]:
            continue
        n_q = index.df.get(term, 0)
        idf = math.log(1 + (index.N - n_q + 0.5) / (n_q + 0.5))
        f = index.tf[doc_i][term]
        denom = f + k1 * (1 - b + b * dl / index.avgdl)
        score += idf * (f * (k1 + 1) / denom) * qtf
    return score


def format_rag_block(chunks: list[tuple[str, str]]) -> str:
    if not chunks:
        return ""
    parts = ["## Retrieved HLS documentation (UG1399)", ""]
    for cid, text in chunks:
        parts.append(f"### chunk {cid}")
        parts.append(text.strip())
        parts.append("")
    return "\n".join(parts).rstrip() + "\n"


def retrieve(index: RagIndex, query: str, top_k: int = DEFAULT_TOP_K) -> str:
    q_toks = tokenize(query)
    if not q_toks or index.N == 0:
        return ""
    qtf: dict[str, int] = {}
    for t in q_toks:
        qtf[t] = qtf.get(t, 0) + 1
    scored = [(_bm25_score(index, qtf, i), i) for i in range(index.N)]
    scored.sort(reverse=True)
    picked: list[tuple[str, str]] = []
    for score, i in scored[: max(1, top_k)]:
        if score <= 0 and picked:
            break
        picked.append((index.chunk_ids[i], index.texts[i]))
    return format_rag_block(picked)


def should_inject(mode: Optional[str], stage: str) -> bool:
    if mode is None:
        return False
    if mode not in RAG_MODES:
        return False
    if stage == "flashopt":
        return mode in ("flashopt", "both", "everywhere")
    if stage == "repair":
        return mode in ("repair", "both", "everywhere")
    if stage == "dataflow":
        return mode == "everywhere"
    return False


def _env_truthy(name: str) -> bool:
    return os.getenv(name, "").strip().lower() in ("1", "true", "yes", "on")


def rag_config_from_env(
    *,
    enabled: Optional[bool] = None,
    mode: Optional[str] = None,
    corpus_dir: Optional[str | Path] = None,
    top_k: Optional[int] = None,
) -> RagConfig:
    en = bool(enabled) if enabled is not None else _env_truthy("C2HLS_RAG")
    raw_mode = mode if mode is not None else os.getenv("C2HLS_RAG_MODE", "").strip()
    resolved_mode: Optional[RagMode] = None
    if en:
        if not raw_mode:
            resolved_mode = "flashopt"
        elif raw_mode in RAG_MODES:
            resolved_mode = raw_mode  # type: ignore[assignment]
        else:
            raise ValueError(f"invalid C2HLS_RAG_MODE/rag-mode: {raw_mode!r}; choose from {RAG_MODES}")
    elif raw_mode:
        raise ValueError("--rag-mode / C2HLS_RAG_MODE requires --rag / C2HLS_RAG=1")
    cdir = Path(
        corpus_dir
        or os.getenv("C2HLS_RAG_CORPUS", "").strip()
        or DEFAULT_CORPUS
    )
    tk = top_k if top_k is not None else int(os.getenv("C2HLS_RAG_TOP_K", str(DEFAULT_TOP_K)) or DEFAULT_TOP_K)
    return RagConfig(enabled=en, mode=resolved_mode, corpus_dir=cdir, top_k=tk)


_INDEX_CACHE: dict[str, RagIndex] = {}


def get_index(corpus_dir: Path) -> RagIndex:
    key = str(Path(corpus_dir).resolve())
    if key not in _INDEX_CACHE:
        _INDEX_CACHE[key] = load_index(Path(corpus_dir))
    return _INDEX_CACHE[key]


def retrieve_for_stage(cfg: RagConfig, stage: str, query: str) -> str:
    if not cfg.enabled or not should_inject(cfg.mode, stage):
        return ""
    idx = get_index(cfg.corpus_dir)
    return retrieve(idx, query, top_k=cfg.top_k)
```

```python
#!/usr/bin/env python3
"""Build bundled UG1399 RAG index for c2hls (chunk 1000 / overlap 200, BM25-ready chunks.jsonl)."""
from __future__ import annotations

import argparse
import hashlib
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path

REPO = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(REPO))

from c2hls_rag import CHUNK_SIZE, OVERLAP, chunk_text  # noqa: E402


def read_source(path: Path) -> str:
    raw = path.read_bytes()
    if path.suffix.lower() == ".pdf":
        try:
            from pypdf import PdfReader  # optional
        except ImportError as e:
            raise SystemExit(
                "PDF support requires pypdf. pip install pypdf  OR pass .txt/.html"
            ) from e
        reader = PdfReader(str(path))
        return "\n".join((p.extract_text() or "") for p in reader.pages)
    text = raw.decode("utf-8", errors="replace")
    if path.suffix.lower() in {".html", ".htm"}:
        text = re.sub(r"(?is)<script.*?>.*?</script>", " ", text)
        text = re.sub(r"(?is)<style.*?>.*?</style>", " ", text)
        text = re.sub(r"(?s)<[^>]+>", " ", text)
        text = re.sub(r"\s+", " ", text)
    return text


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--source", type=Path, required=True)
    ap.add_argument("--out", type=Path, default=REPO / "artifacts" / "rag" / "ug1399")
    ap.add_argument("--chunk-size", type=int, default=CHUNK_SIZE)
    ap.add_argument("--overlap", type=int, default=OVERLAP)
    args = ap.parse_args()
    if not args.source.is_file():
        raise SystemExit(f"source not found: {args.source}")
    text = read_source(args.source)
    sha = hashlib.sha256(args.source.read_bytes()).hexdigest()
    chunks = chunk_text(text, chunk_size=args.chunk_size, overlap=args.overlap)
    args.out.mkdir(parents=True, exist_ok=True)
    chunks_path = args.out / "chunks.jsonl"
    with chunks_path.open("w", encoding="utf-8") as f:
        for i, c in enumerate(chunks):
            f.write(json.dumps({"id": f"c{i}", "text": c}, ensure_ascii=False) + "\n")
    meta = {
        "engine": "bm25",
        "chunk_size": args.chunk_size,
        "overlap": args.overlap,
        "source": str(args.source.resolve()),
        "source_sha256": sha,
        "n_chunks": len(chunks),
        "built_at": datetime.now(timezone.utc).isoformat(),
    }
    (args.out / "index_meta.json").write_text(json.dumps(meta, indent=2) + "\n", encoding="utf-8")
    print(f"wrote {len(chunks)} chunks -> {args.out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
